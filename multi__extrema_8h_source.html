50 unsigned int max_iter = OPTIMIZATION_MINGRAD_ITER
51 ) {
52
53 ReturnVector x = guess;
54
55 if(gamma <= 0) {
56 TH_MATH_ERROR("multi_minimize_grad", gamma, MathError::InvalidArgument);
57 return algebra::vec_error(x);
58 }
59
60 ReturnVector grad;
61 unsigned int iter = 0;
62
63 do {
64
65 grad = autodiff::gradient(f, x);
66 x -= gamma * grad;
67 iter++;
68
69 } while(grad.norm() > tolerance && iter <= max_iter);
70
71 if(iter > max_iter) {
72 TH_MATH_ERROR("multi_minimize_grad", iter, MathError::NoConvergence);
73 return algebra::vec_error(x);
74 }

76 return x;

149 inline ReturnVector multi_minimize_lingrad(
150 ObjectiveFunction f,
151 Vector guess,
152 real tolerance = OPTIMIZATION_MINGRAD_TOLERANCE,
153 unsigned int max_iter = OPTIMIZATION_MINGRAD_ITER
154 ) {
155
156 ReturnVector x = guess;
157 ReturnVector grad;
158 unsigned int iter = 0;
159 constexpr size_t N = ReturnVector::size_argument;
160
161 do {
162
163 grad = autodiff::gradient(f, x);
164
165 // Minimize f(x + gamma * gradient) in [-1, 0]
166 // using Golden Section extrema search
167 auto result = minimize_golden(
168 [f, x, grad](real gamma){
169 return f(multidual<N>::make_argument(x + gamma * grad)).Re();
170 }, -1, 0);
171
172 // Fallback value
173 real gamma = OPTIMIZATION_MINGRAD_GAMMA;
174 if (result.converged())
175 gamma = result.value;
176
177 x += gamma * grad;
178 iter++;
179
180 } while(grad.norm() > tolerance && iter < max_iter);
181
182 if(iter > max_iter) {
183 TH_MATH_ERROR("multi_minimize_lingrad", iter, MathError::NoConvergence);
184 return algebra::vec_error(x);
185 }
186
187 return x;
188 }
