28 inline real uniform(real a, real b, PRNG& g, uint64_t prec = PSEUDORANDOM_PREC) {
29
30 // Generate a uniform random real number in [0, 1]
31 real x = (g() % prec) / static_cast<real>(prec);
32
33 // Transform to target interval
34 return a + (b - a) * x;
35 }

88 template<typename PRNG>
89 inline real trycatch(
90 stat_function f,
91 const vec<real>& theta,
92 real x1, real x2,
93 real y1, real y2, PRNG& g,
94 unsigned int max_iter = STATISTICS_TRYANDCATCH_ITER
95 ) {
96
97 real x;
98 real y;
99
100 unsigned int iter = 0;
101
102 do {
103 x = random::uniform(x1, x2, g);
104 y = random::uniform(y1, y2, g);
105 iter++;
106 } while(y > f(x, theta) && iter <= max_iter);
107
108 if(iter > max_iter) {
109 TH_MATH_ERROR("random::trycatch", iter, MathError::NoConvergence);
110 return nan();
111 }
112
113 return x;
114 }

127 template<typename PRNG>
128 inline real rejectsamp(
129 stat_function f, const vec<real>& theta,
130 real_function p, real_function Pinv,
131 PRNG& g, unsigned int max_tries = 100) {
132
133 for (unsigned int i = 0; i < max_tries; ++i) {
134
135 // Generate a random number following
136 // the p(x) probability distribution
137 // by the inverse cumulative distribution function
138 const real u_1 = random::uniform(0, 1, g);
139 const real x_p = Pinv(u_1);
140
141 const real u_2 = random::uniform(0, 1, g);
142
143 // Accept the sample if f(x_p)/g(x_p) > u_2
144 if(u_2 * p(x_p) < f(x_p, theta))
145 return x_p;
146 }
147
148 TH_MATH_ERROR("random::rejectsamp", max_tries, MathError::NoConvergence);
149 return nan();
150 }

158 template<typename PRNG>
159 inline real gaussian_polar(real mean, real sigma, PRNG& g) {
160
161 static real spare;
162 static bool has_spare = false;
163
164 if(has_spare) {
165 has_spare = false;
166 return mean + spare * sigma;
167 }
168
169 real x, y, s;
170
171 // Generate a random point inside the unit circle
172 do {
173
174 x = random::uniform(-1, 1, g);
175 y = random::uniform(-1, 1, g);
176 s = square(x) + square(y);
177
178 } while(s >= 1 || s <= MACH_EPSILON);
179
180 // Project the point
181 s = sqrt(-2 * ln(s) / s);
182
183 // Keep the second generated value for future calls
184 spare = y * s;
185 has_spare = true;
186
187 return mean + sigma * x * s;
188 }

196 template<typename PRNG>
197 inline real gaussian_boxmuller(real mean, real sigma, PRNG& g) {
198
199 static real spare;
200 static bool has_spare = false;
201
202 if(has_spare) {
203 has_spare = false;
204 return mean + spare * sigma;
205 }
206
207 // Generate a random point inside the unit circle
208
209 const real x = random::uniform(0, 1, g);
210 const real y = random::uniform(0, 1, g);
211
212 const real x_transf = sqrt(-2 * ln(x));
213
214 const real u = x_transf * cos(TAU * y);
215 const real v = x_transf * sin(TAU * y);
216
217 spare = v;
218 has_spare = true;
219
220 return mean + sigma * u;
221 }

235 template<typename PRNG>
236 inline real gaussian_clt(real mean, real sigma, PRNG& g) {
237
238 // Fixed N = 12
239 constexpr unsigned int N = 12;
240
241 real s = 0;
242 for (unsigned int i = 0; i < N; ++i)
243 s += random::uniform(-1, 1, g);
244
245 // f(u) = 1/2 (in [-1, 1])
246 // E[u] = 0
247 // sqrt(V[u]) = 1 / sqrt(3N) = 1 / 6
248
249 return mean + (s / static_cast<real>(N)) * sigma * 6;
250 }

563 template <
564 typename RealFunction,
565 typename PRNG1,
566 typename PRNG2
567 >
568 inline real metropolis(
569 RealFunction pdf, PdfSampler<PRNG1>& g,
570 real x0, PRNG2& rnd, unsigned int depth = STATISTICS_METROPOLIS_DEPTH) {
571
572 real current = x0, next;
573
574 for(unsigned int i = 0; i < depth; i++) {
575
576 // Computes the next step
577 next = current + g();
578
579 // Checks acceptance rate
580 if(random::uniform(0, 1, rnd) * pdf(current) <= pdf(next))
581 current = next;
582 }
583
584 return current;
585 }
