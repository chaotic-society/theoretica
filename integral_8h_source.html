23 template<typename Field = real>
24 inline polynomial<Field> integral(const polynomial<Field>& p) {
25
26 if (p.size() == 0)
27 return polynomial<Field>({});
28
29 polynomial<Field> P;
30 P.resize(p.size() + 1);
31
32 if (P.size() != p.size() + 1) {
33 TH_MATH_ERROR("integral(polynomial)", P.size(), MathError::ImpossibleOperation);
34 return polynomial<Field>(nan());
35 }
36
37 P[0] = Field(0.0);
38 for (unsigned int i = 0; i < p.size(); ++i)
39 P[i + 1] = p[i] / Field(i + 1);
40
41 return P;
42 }

53 inline real integral(const polynomial<real>& p, real a, real b) {
54
55 real P_a = 0.0;
56 real P_b = 0.0;
57
58 for (unsigned int i = 0; i < p.size(); ++i) {
59
60 const unsigned int pos = p.size() - i - 1;
61 P_a = a * (p[pos] / (pos + 1) + P_a);
62 P_b = b * (p[pos] / (pos + 1) + P_b);
63 }
64
65 return P_b - P_a;
66 }

77 template<typename RealFunction>
78 inline real integral_midpoint(RealFunction f, real a, real b,
79 unsigned int steps = CALCULUS_INTEGRAL_STEPS) {
80
81 if(steps == 0) {
82 TH_MATH_ERROR("integral_midpoint", steps, MathError::DivByZero);
83 return nan();
84 }
85
86 const real dx = (b - a) / steps;
87 real res = 0;
88
89 for (unsigned int i = 0; i < steps; ++i)
90 res += f(a + (i + 0.5) * dx);
91
92 return res * dx;
93 }

104 template<typename RealFunction>
105 inline real integral_trapezoid(RealFunction f, real a, real b,
106 unsigned int steps = CALCULUS_INTEGRAL_STEPS) {
107
108 if(steps == 0) {
109 TH_MATH_ERROR("integral_trapezoid", steps, MathError::DivByZero);
110 return nan();
111 }
112
113 const real dx = (b - a) / steps;
114 real res = 0;
115
116 res += 0.5 * (f(a) + f(b));
117
118 for (unsigned int i = 1; i < steps; ++i)
119 res += f(a + i * dx);
120
121 return res * dx;
122 }

133 template<typename RealFunction>
134 inline real integral_simpson(RealFunction f, real a, real b,
135 unsigned int steps = CALCULUS_INTEGRAL_STEPS) {
136
137 if (steps == 0) {
138 TH_MATH_ERROR("integral_simpson", steps, MathError::DivByZero);
139 return nan();
140 }
141
142 if (steps % 2 != 0) {
143 TH_MATH_ERROR("integral_simpson", steps, MathError::InvalidArgument);
144 return nan();
145 }
146
147 const real dx = (b - a) / (real) steps;
148 real res = 0;
149
150 // Sum terms by order of magnitude supposing that
151 // f stays at the same order inside the interval,
152 // to alleviate truncation errors
153
154 res += f(a) + f(b);
155
156 for (unsigned int i = 2; i < steps; i += 2)
157 res += 2 * f(a + i * dx);
158
159 for (unsigned int i = 1; i < steps; i += 2)
160 res += 4 * f(a + i * dx);
161
162 return res * dx / 3.0;
163 }

175 template<typename RealFunction>
176 inline real integral_romberg(
177 RealFunction f,
178 real a, real b,
179 unsigned int iter = 8) {
180
181 const unsigned int MAX_ROMBERG_ITER = 16;
182 real T[MAX_ROMBERG_ITER][MAX_ROMBERG_ITER];
183
184 if (iter > MAX_ROMBERG_ITER) {
185 TH_MATH_ERROR("integral_romberg", iter, MathError::InvalidArgument);
186 return nan();
187 }
188
189 T[0][0] = (f(a) + f(b)) * (b - a) / 2.0;
190
191 for (unsigned int j = 1; j < iter; ++j) {
192
193 // Composite Trapezoidal Rule
194 T[j][0] = integral_trapezoid(f, a, b, 1 << j);
195
196 // Richardson extrapolation
197 for (unsigned int k = 1; k <= j; ++k) {
198 const uint64_t coeff = uint64_t(1) << (2 * k);
199 T[j][k] = (coeff * T[j][k - 1] - T[j - 1][k - 1]) / (coeff - 1);
200 }
201 }
202
203 // Return the best approximation
204 return T[iter - 1][iter - 1];
205 }

216 template<typename RealFunction>
217 inline real integral_romberg_tol(
218 RealFunction f,
219 real a, real b,
220 real tolerance = CALCULUS_INTEGRAL_TOL) {
221
222 const unsigned int MAX_ROMBERG_ITER = 16;
223 real T[MAX_ROMBERG_ITER][MAX_ROMBERG_ITER];
224
225 T[0][0] = (f(a) + f(b)) * (b - a) / 2.0;
226
227 for (unsigned int j = 1; j < MAX_ROMBERG_ITER; ++j) {
228
229 // Composite Trapezoidal Rule
230 T[j][0] = integral_trapezoid(f, a, b, 1 << j);
231
232 // Richardson extrapolation
233 for (unsigned int k = 1; k <= j; ++k) {
234 const uint64_t coeff = uint64_t(1) << (2 * k);
235 T[j][k] = (coeff * T[j][k - 1] - T[j - 1][k - 1]) / (coeff - 1);
236 }
237
238 // Stop the algorithm when the desired precision has been reached
239 if(abs(T[j][j] - T[j - 1][j - 1]) < tolerance)
240 return T[j][j];
241 }
242
243 // Return the best approximation
244 return T[MAX_ROMBERG_ITER - 1][MAX_ROMBERG_ITER - 1];
245 }

320 template<typename RealFunction>
321 inline real integral_legendre(
322 RealFunction f, real a, real b, real* x, real* w, unsigned int n) {
323
324 const real mean = (b + a) / 2.0;
325 const real halfdiff = (b - a) / 2.0;
326
327 real res = 0;
328
329 for (int i = n - 1; i >= 0; --i)
330 res += w[i] * f(halfdiff * x[i] + mean);
331
332 return res * halfdiff;
333 }

431 inline real integral_laguerre(
432 RealFunction f, real a, real b, const std::vector<real>& x) {
433
434 const std::vector<real> weights = laguerre_weights(x);
435
436 const real exp_a = exp(-a);
437 const real exp_b = exp(-b);
438
439 real res = 0;
440
441 for (int i = x.size() - 1; i >= 0; --i)
442 res += weights[i] * (exp_a * f(x[i] + a) - exp_b * f(x[i] + b));
443
444 return res;
445 }

523 inline real integral_inf_riemann(
524 real_function f, real a, real step_sz = 1,
525 real tol = CALCULUS_INTEGRAL_TOL, unsigned int max_iter = 100) {
526
527 // Current lower extreme of the interval
528 real x_n = a + step_sz;
529
530 // Total integral sum
531 real sum = integral_romberg_tol(f, a, x_n, tol);
532
533 // Variation between steps
534 real delta = inf();
535
536 // Number of steps performed
537 unsigned int i = 0;
538
539 while(abs(delta) > tol && i <= max_iter) {
540
541 delta = integral_romberg_tol(f, x_n, x_n + step_sz, tol);
542 sum += delta;
543 x_n += step_sz;
544 i++;
545 }
546
547 if(i >= max_iter) {
548 TH_MATH_ERROR("integral_inf_riemann", i, MathError::NoConvergence);
549 return nan();
550 }
551
552 return sum;
553 }
