180 template<typename Vector, typename PRNG>
181 inline Vector& shuffle(Vector& v, PRNG& g, unsigned int rounds) {
182
183 if(v.size() == 0)
184 TH_MATH_ERROR("shuffle", v.size(), MathError::InvalidArgument);
185
186 for (unsigned int i = 0; i < rounds; ++i) {
187
188 // Generate two random indices
189 const size_t index1 = g() % v.size();
190 const size_t index2 = g() % v.size();
191
192 // Exchange the two values at the random indices
193 std::swap(v[index1], v[index2]);
194 }
195
196 return v;
197 }
