32 inline stoch_result<real> integral_mc(
33 RealFunction f, real a, real b,
34 PRNG& g, size_t N = 1'000'000
35 ) {
36
37 // Statistics are computed online using Welford's method.
38 stats::RunningMoments2<real> stats;
39
40 // Pre-generate seeds for all threads to avoid contention
41 int num_threads;
42 #pragma omp parallel
43 {
44 #pragma omp single
45 num_threads = omp_get_num_threads();
46 }
47
48 vec<uint64_t> seeds (num_threads);
49 for (int i = 0; i < num_threads; ++i)
50 seeds[i] = g();
51
52 #pragma omp parallel
53 {
54 int id = omp_get_thread_num();
55 stats::RunningMoments2<real> local_stats;
56 PRNG local_g = PRNG(seeds[id]);
57
58 #pragma omp for nowait
59 for (unsigned int i = 0; i < N; ++i) {
60 const real x = random::uniform(a, b, local_g);
61 local_stats.insert(f(x));
62 }
63
64 #pragma omp critical
65 {
66 // Combine statistics accross threads using Chan's formula
67 stats.combine(local_stats);
68 }
69 }
70
71 const real volume = (b - a);
72

74 volume * stats.mean(),

109 #pragma omp single

114 for (int i = 0; i < num_threads; ++i)
115 seeds[i] = g();
116

118 {

120 stats::RunningMoments2<ReturnType> local_stats;
121 PRNG local_g = PRNG(seeds[id]);
122
123 Vector v;
124 v.resize(dim);
125
126 #pragma omp for nowait
127 for (unsigned int i = 0; i < N; ++i) {
128
129 for (unsigned int k = 0; k < v.size(); ++k)
130 v[k] = random::uniform(extremes[k][0], extremes[k][1], local_g);
131
132 local_stats.insert(f(v));
133 }
134
135 #pragma omp critical
136 {
137 // Combine statistics accross threads using Chan's formula
138 stats.combine(local_stats);
139 }
140 }
141

143 for (unsigned int i = 0; i < dim; ++i)
144 volume *= (extremes[i][1] - extremes[i][0]);
145
146 return stoch_result<ReturnType>{
147 volume * stats.mean(),
148 volume * sqrt(stats.variance() * (1.0 / stats.number())),

164 template<typename RealFunction, typename PRNG>
165 inline stoch_result<real> integral_qmc(
166 RealFunction f, real a, real b,
167 PRNG& g, size_t N = 1'000'000, size_t N_shift = 10
168 ) {
169
170 stats::RunningMoments2<real> stats;
171
172 // Pre-generate shifts to avoid contention during parallel region
173 vec<real> shifts (N_shift);
174 for (unsigned int i = 0; i < N_shift; ++i)
175 shifts[i] = random::uniform(0, 1, g);
176
177 #pragma omp parallel for
178 for (unsigned int j = 0; j < N_shift; ++j) {
179
180 // Random shift for the Weyl sequence
181 real shift = shifts[j];
182
183 // Distribute points over shifts
184 const size_t size = N / N_shift;
185 real sum_y = 0;
186
187 for (unsigned int i = 0; i < size; ++i) {
188
189 const real x = a + fract(i * INVPHI + shift) * (b - a);
190 sum_y += f(x);
191 }
192
193 #pragma omp critical
194 {
195 const real estimate = (b - a) * sum_y / size;
196 stats.insert(estimate);
197 }
198 }
199
200 return stoch_result<real>{
201 stats.mean(),
202 sqrt(stats.variance()),
203 N
204 };
205 }
