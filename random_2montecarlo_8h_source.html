33 template<typename RealFunction, typename PRNG>
34 inline stoch_result<real> integral_mc(
35 RealFunction f, real a, real b,
36 PRNG& g, size_t N = 1'000'000
37 ) {
38
39 // Parallel dispatch
40 #ifdef _OPENMP
41 if (N >= 10'000)
42 return parallel::integral_mc(f, a, b, g, N);
43 #endif
44
45 // Statistics are computed online using Welford's method.
46 stats::RunningMoments2<real> stats;
47 for (unsigned int i = 0; i < N; ++i) {
48 const real x = random::uniform(a, b, g);
49 stats.insert(f(x));
50 }
51
52 return stoch_result<real>{
53 stats.mean() * (b - a),
54 sqrt(stats.variance() * ((b - a) * (b - a) / stats.number())),
55 stats.number()
56 };
57 }

66 template <
67 typename Function,
68 typename DomainVector = vec<vec2>,
69 typename PRNG,
70 typename ReturnType = return_type_t<Function>
71 >
72 inline stoch_result<ReturnType> integral_mc(
73 Function f, DomainVector extremes,
74 PRNG& g, size_t N = 1'000'000
75 ) {
76
77 using Vector = typename _internal::func_helper<Function>::first_arg_type;
78 const size_t dim = extremes.size();
79
80
81 // Parallel dispatch
82 #ifdef _OPENMP
83 if (N * dim >= 10'000)
84 return parallel::integral_mc(f, extremes, g, N);
85 #endif
86
87
88 // Statistics are computed online using Welford's method.
89 stats::RunningMoments2<ReturnType> stats;
90
91 // Initialize sum by evaluating a first point
92 Vector v;
93 v.resize(dim);
94
95 // Sample the function at random points in the integration region
96 for (unsigned int i = 0; i < N; ++i) {
97
98 for (unsigned int k = 0; k < v.size(); ++k)
99 v[k] = random::uniform(extremes[k][0], extremes[k][1], g);
100
101 stats.insert(f(v));
102 }
103
104 // Volume of the integral domain
105 real volume = 1;
106 for (unsigned int i = 0; i < dim; ++i)
107 volume *= (extremes[i][1] - extremes[i][0]);
108
109
110 // Return result as a stochastic result
111 return stoch_result<ReturnType>{
112 stats.mean() * volume,
113 volume * sqrt(stats.variance() * (1.0 / stats.number())),
114 stats.number()
115 };
116 }

129 template<typename RealFunction, typename PRNG>
130 inline stoch_result<real> integral_qmc(
131 RealFunction f, real a, real b,
132 PRNG& g, size_t N = 1'000'000, size_t M = 10
133 ) {
134
135 // Parallel dispatch
136 #ifdef _OPENMP
137 if (N >= 1'000'000)
138 return parallel::integral_qmc(f, a, b, g, N, M);
139 #endif
140
141
142 stats::RunningMoments2<real> stats;
143 for (unsigned int run = 0; run < M; ++run) {
144
145 // Random shift for the Weyl sequence
146 const real shift = random::uniform(0, 1, g);
147
148 // Distribute points over shifts
149 const size_t size = N / M;
150
151 real sum_y = 0;
152 for (unsigned int i = 0; i < size; ++i) {
153
154 const real x = a + fract(i * INVPHI + shift) * (b - a);
155 sum_y += f(x);
156 }
157
158 const real estimate = (b - a) * sum_y / size;
159 stats.insert(estimate);
160 }
161
162 return stoch_result<real>{
163 stats.mean(),
164 sqrt(stats.variance()),
165 N
166 };
167 }

180 template <
181 typename Function,
182 typename DomainVector = vec<vec2>,
183 typename PRNG,
184 typename ReturnType = return_type_t<Function>,
185 typename AlphaVector = vec<real>
186 >
187 inline stoch_result<ReturnType> integral_qmc(
188 Function f, DomainVector extremes,
189 PRNG& g, size_t N = 1'000'000, size_t M = 10,
190 AlphaVector alphas = {}
191 ) {
192
193 using Vector = typename _internal::func_helper<Function>::first_arg_type;
194 const size_t dim = extremes.size();
195
196 // If not provided, compute the optimal
197 // parameters for the Weyl sequence.
198 if (alphas.size() == 0) {
199
200 alphas.resize(dim);
201
202 // Find the only positive root of the polynomial: x^s+1 - x - 1 = 0
203 const real base = 1.0 / root_bisect([=](real x) {
204 return pow(x, dim + 1) - x - 1;
205 }, 0.0, 2.0);
206
207 // Optimal parameters are powers of the base
208 real alpha = base;
209 for (unsigned int i = 0; i < dim; ++i) {
210 alphas[i] = alpha;
211 alpha *= base;
212 }
213 }
214
215 // Volume of the integral domain
216 real volume = 1.0;
217 for (unsigned int i = 0; i < dim; ++i)
218 volume *= (extremes[i][1] - extremes[i][0]);
219
220
221 stats::RunningMoments2<ReturnType> stats;
222 for (unsigned int run = 0; run < M; ++run) {
223
224 // Random shift for the Weyl sequence
225 Vector shift;
226 shift.resize(dim);
227 for (unsigned int i = 0; i < shift.size(); ++i)
228 shift[i] = random::uniform(0, 1, g);
229
230 // Distribute points over shifts
231 const size_t size = N / M;
232
233 Vector x;
234 x.resize(dim);
235
236 ReturnType sum_y = 0;
237 for (unsigned int i = 0; i < size; ++i) {
238
239 for (unsigned int k = 0; k < x.size(); ++k) {
240 const real xi = fract(i * alphas[k] + shift[k]);
241 x[k] = extremes[k][0] + xi * (extremes[k][1] - extremes[k][0]);
242 }
243
244 sum_y += f(x);
245 }
246
247 stats.insert(volume * sum_y / size);
248 }
249
250 return stoch_result<ReturnType>{
251 stats.mean(),
252 sqrt(stats.variance()),
253 N
254 };
255 }
